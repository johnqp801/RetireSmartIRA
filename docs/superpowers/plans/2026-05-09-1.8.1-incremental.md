```swift
//
//  TabPurposeChip.swift
//  RetireSmartIRA
//
//  Status chip indicating whether a tab is for data entry ("Inputs")
//  or for results/analysis ("Analysis"). Per Ron Park beta feedback.
//

import SwiftUI

enum TabPurpose {
    case inputs
    case analysis
    
    var label: String {
        switch self {
        case .inputs: return "Inputs"
        case .analysis: return "Analysis"
        }
    }
    
    var icon: String {
        switch self {
        case .inputs: return "square.and.pencil"
        case .analysis: return "chart.bar.doc.horizontal"
        }
    }
    
    var color: Color {
        switch self {
        case .inputs: return .blue
        case .analysis: return .green
        }
    }
}

struct TabPurposeChip: View {
    let purpose: TabPurpose
    
    var body: some View {
        HStack(spacing: 4) {
            Image(systemName: purpose.icon)
                .font(.caption2)
            Text(purpose.label)
                .font(.caption.weight(.medium))
        }
        .foregroundStyle(purpose.color)
        .padding(.horizontal, 8)
        .padding(.vertical, 3)
        .background(purpose.color.opacity(0.12))
        .clipShape(Capsule())
        .overlay(Capsule().stroke(purpose.color.opacity(0.25), lineWidth: 0.5))
    }
}
```

```swift
//  RetireSmartIRA
//
//  Live ACA cliff indicator displayed near the Roth conversion slider.
//  Shows current headroom to the cliff with green/yellow/red color states.
//  Per Ron Park beta feedback (Item #10 in 1.8.1 spec).

import SwiftUI

enum ACACliffHeadroomState: Equatable {
    case green       // > $20K headroom
    case yellow      // $5K to $20K
    case red         // $0 to $5K (within buffer)
    case crossed     // < $0 (over cliff)
    
    var color: Color {
        switch self {
        case .green: return .green
        case .yellow: return .orange
        case .red: return .red
        case .crossed: return .red
        }
    }
    
    var label: String {
        switch self {
        case .green: return "Clear of cliff"
        case .yellow: return "Approaching cliff"
        case .red: return "Near cliff"
        case .crossed: return "Cliff crossed"
        }
    }
}

struct ACACliffIndicator: View {
    let cliffThreshold: Double      // household income at which subsidy is lost
    let projectedMAGI: Double       // current/projected ACAMAGI
    let lostSubsidyEstimate: Double // estimated annual subsidy if crossed
    
    private var headroom: Double {
        cliffThreshold - projectedMAGI
    }
    
    private var state: ACACliffHeadroomState {
        Self.headroomState(headroom: headroom)
    }
    
    static func headroomState(headroom: Double) -> ACACliffHeadroomState {
        if headroom < 0 { return .crossed }
        if headroom < 5_000 { return .red }
        if headroom < 20_000 { return .yellow }
        return .green
    }
    
    var body: some View {
        VStack(alignment: .leading, spacing: 4) {
            HStack(spacing: 6) {
                Circle().fill(state.color).frame(width: 8, height: 8)
                Text(state.label)
                    .font(.caption.weight(.semibold))
                    .foregroundStyle(state.color)
                Spacer()
            }
            
            HStack {
                Text("ACA cliff: \(formatDollars(cliffThreshold))")
                    .font(.caption)
                    .foregroundStyle(.secondary)
                Spacer()
                Text(state == .crossed
                    ? "Over by \(formatDollars(abs(headroom)))"
                    : "Headroom: \(formatDollars(headroom))")
                    .font(.caption.weight(.medium))
                    .foregroundStyle(state.color)
                    .monospacedDigit()
            }
            
            if state == .crossed {
                Text("⚠ ACA subsidy lost — costs ~\(formatDollars(lostSubsidyEstimate))/yr")
                    .font(.caption2)
                    .foregroundStyle(.red)
                    .padding(.top, 2)
            }
        }
        .padding(8)
        .background(state.color.opacity(0.06))
        .clipShape(RoundedRectangle(cornerRadius: 6))
        .overlay(
            RoundedRectangle(cornerRadius: 6)
                .stroke(state.color.opacity(0.25), lineWidth: 0.5)
        )
    }
    
    private func formatDollars(_ value: Double) -> String {
        let abs = abs(value)
        if abs >= 1_000_000 {
            return String(format: "$%.1fM", value / 1_000_000)
        } else if abs >= 1_000 {
            return String(format: "$%.0fK", value / 1_000)
        } else {
            return String(format: "$%.0f", value)
        }
    }
}
```

```swift
//
//  MilitaryRetirementExemption.swift
//  RetireSmartIRA
//
//  State-by-state exemption table for military retirement pay.
//  Per Tim's beta feedback (Item #18 in 1.8.1 spec).
//
//  Sources:
//  - Military.com state tax exemption guides (2025-2026)
//  - DFAS state tax withholding policies
//  - Individual state department of revenue publications
//
//  IMPORTANT: State tax laws change. Users should verify their state's
//  specific rules with a tax professional. This table represents a
//  best-effort 2026 snapshot.
//

import Foundation

enum MilitaryRetirementExemptionLevel: Equatable {
    case fullyExempt
    case partiallyExempt(percentTaxable: Double, ageThreshold: Int?)  // e.g., 50% taxable above age 60
    case fullyTaxable
    case noStateIncomeTax
}

enum MilitaryRetirementExemption {
    
    static func exemption(for stateCode: String, age: Int) -> MilitaryRetirementExemptionLevel {
        switch stateCode.uppercased() {
        // No state income tax
        case "AK", "FL", "NV", "SD", "TX", "WA", "WY":
            return .noStateIncomeTax
        case "NH":
            // NH taxes only interest/dividends (not retirement income)
            return .noStateIncomeTax
        case "TN":
            return .noStateIncomeTax
        
        // Fully exempt — military retirement excluded from state taxable income
        case "AL", "AR", "CT", "HI", "IL", "KS", "LA", "ME", "MA", "MI",
             "MN", "MS", "MO", "NJ", "NC", "ND", "OH", "OK", "PA", "SC",
             "WI", "AZ", "IN", "NE", "UT", "WV":
            return .fullyExempt
        
        // Conditionally exempt — full exemption above age threshold
        case "IA":
            return age >= 55 ? .fullyExempt : .fullyTaxable
        
        // Partially exempt
        case "NY":
            // NY: military retirement is fully exempt
            return .fullyExempt
        case "NM":
            // NM: partial exemption based on income; modeled as 0.5 taxable for simplicity
            return .partiallyExempt(percentTaxable: 0.5, ageThreshold: nil)
        case "OR":
            // OR: $6K subtraction; modeled as fully taxable for simplicity at typical pension levels
            return .fullyTaxable  // simplification, $6K subtraction is small
        
        // Fully taxable
        case "CA", "DC", "GA", "ID", "KY", "MD", "MT", "RI", "VT", "VA", "DE":
            return .fullyTaxable
        
        // Default: assume fully taxable (conservative) for unknown / not-yet-mapped states
        default:
            return .fullyTaxable
        }
    }
    
    /// Returns the dollar amount of military retirement income that's taxable for state purposes,
    /// given the gross amount and the user's state and age.
    static func stateTaxableAmount(gross: Double, stateCode: String, age: Int) -> Double {
        switch exemption(for: stateCode, age: age) {
        case .fullyExempt, .noStateIncomeTax:
            return 0
        case .fullyTaxable:
            return gross
        case .partiallyExempt(let percentTaxable, let ageThreshold):
            if let threshold = ageThreshold, age < threshold {
                return gross  // not yet eligible for partial exemption
            }
            return gross * percentTaxable
        }
    }
}
```
